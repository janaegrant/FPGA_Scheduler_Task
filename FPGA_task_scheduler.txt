module final_3201P(
    input  wire MAX10_CLK1_50,
    input  wire [1:0] KEY,
    input  wire [9:0] SW,        // SW[0]=SOS, SW[1]=WARNING
    output reg  [9:0] LEDR,
    output wire [6:0] HEX0,
    output wire [6:0] HEX1,
    output wire [6:0] HEX5,
    output wire [6:0] HEX4,
    output wire [6:0] HEX3
);

    wire reset_n = KEY[0];
    wire tick;
    wire sos_mode  = SW[0];
    wire warn_mode = SW[1];

    // combined pause signal — freezes everything during either interrupt
    wire pause = sos_mode || warn_mode;

    //----------------------------------------
    // CLOCK (1 Hz)
    //----------------------------------------
    clock clkdiv (
        .clk     (MAX10_CLK1_50),
        .reset_n (reset_n),
        .tick    (tick)
    );

    //----------------------------------------
    // TIMER (freezes during any interrupt)
    //----------------------------------------
    reg [5:0] seconds;
    wire [3:0] ones;
    wire [3:0] tens;

    assign ones = seconds % 6'd10;
    assign tens = seconds / 6'd10;

    always @(posedge MAX10_CLK1_50 or negedge reset_n) begin
        if (!reset_n)
            seconds <= 6'd0;
        else if (tick && !pause)
            seconds <= (seconds == 6'd59) ? 6'd0 : seconds + 6'd1;
    end

    //----------------------------------------
    // TASK SELECTION
    //----------------------------------------
    wire is_blink_second =
        (seconds==6'd10)||(seconds==6'd20)||(seconds==6'd30)||
        (seconds==6'd40)||(seconds==6'd50);

    wire is_task3 = (!is_blink_second &&
        ((seconds <= 6'd9) ||
         (seconds >= 6'd21 && seconds <= 6'd29) ||
         (seconds >= 6'd41 && seconds <= 6'd49)));

    wire is_task4 = (!is_blink_second && !is_task3);

    wire t2_en = is_blink_second && !pause;
    wire t3_en = is_task3        && !pause;
    wire t4_en = is_task4        && !pause;

    //----------------------------------------
    // HEX SEGMENT CONSTANTS
    //----------------------------------------
    localparam [6:0] SEG_BLANK = 7'b1111111;
    localparam [6:0] SEG_S     = 7'b0010010;
    localparam [6:0] SEG_O     = 7'b1000000;
    localparam [6:0] SEG_2     = 7'b0100100;
    localparam [6:0] SEG_3     = 7'b0110000;
    localparam [6:0] SEG_4     = 7'b0011001;

    //----------------------------------------
    // HEX0 and HEX1 — seconds timer, blank during SOS
    //----------------------------------------
    wire [6:0] hex0_normal, hex1_normal;
    sevenseg_driver hex_ones (.value(ones), .seg(hex0_normal));
    sevenseg_driver hex_tens (.value(tens), .seg(hex1_normal));

    assign HEX0 = sos_mode ? SEG_BLANK : hex0_normal;
    assign HEX1 = sos_mode ? SEG_BLANK : hex1_normal;

    //----------------------------------------
    // HEX5/4/3 — task number, or "SOS" during SOS
    //----------------------------------------
    reg [6:0] hex5_reg, hex4_reg, hex3_reg;

    always @(*) begin
        if (sos_mode) begin
            hex5_reg = SEG_S;
            hex4_reg = SEG_O;
            hex3_reg = SEG_S;
        end else if (is_blink_second) begin
            hex5_reg = SEG_2;
            hex4_reg = SEG_BLANK;
            hex3_reg = SEG_BLANK;
        end else if (is_task3) begin
            hex5_reg = SEG_3;
            hex4_reg = SEG_BLANK;
            hex3_reg = SEG_BLANK;
        end else begin
            hex5_reg = SEG_4;
            hex4_reg = SEG_BLANK;
            hex3_reg = SEG_BLANK;
        end
    end

    assign HEX5 = hex5_reg;
    assign HEX4 = hex4_reg;
    assign HEX3 = hex3_reg;

    //----------------------------------------
    // TASKS
    //----------------------------------------
    wire [9:0] t2_LEDRs, t3_LEDRs, t4_LEDRs;

    task2 t2 (
        .clk     (MAX10_CLK1_50),
        .reset_n (reset_n),
        .tick    (tick),
        .enable  (t2_en),
        .seconds (seconds),
        .LEDRs   (t2_LEDRs)
    );

    task3 t3 (
        .clk      (MAX10_CLK1_50),
        .reset_n  (reset_n),
        .tick     (tick),
        .enable   (t3_en),
        .sos_mode (pause),       // freeze on ANY interrupt
        .LEDRs    (t3_LEDRs)
    );

    task4 t4 (
        .clk      (MAX10_CLK1_50),
        .reset_n  (reset_n),
        .tick     (tick),
        .enable   (t4_en),
        .sos_mode (pause),       // freeze on ANY interrupt
        .LEDRs    (t4_LEDRs)
    );

    //----------------------------------------
    // SOS FLASHER (Morse: S=... O=--- S=...)
    // 4 Hz tick: dot = 1 tick on + 1 off, dash = 3 on + 1 off
    //----------------------------------------
    reg [23:0] tick4_cnt;
    reg tick_4hz;

    always @(posedge MAX10_CLK1_50 or negedge reset_n) begin
        if (!reset_n) begin
            tick4_cnt <= 24'd0;
            tick_4hz  <= 1'b0;
        end else begin
            if (tick4_cnt == 24'd12_499_999) begin
                tick4_cnt <= 24'd0;
                tick_4hz  <= 1'b1;
            end else begin
                tick4_cnt <= tick4_cnt + 24'd1;
                tick_4hz  <= 1'b0;
            end
        end
    end

    reg sos_led;
    reg [3:0] sos_step, sos_count;

    always @(posedge MAX10_CLK1_50 or negedge reset_n) begin
        if (!reset_n) begin
            sos_step  <= 4'd0;
            sos_count <= 4'd0;
            sos_led   <= 1'b0;
        end else if (!sos_mode) begin
            sos_step  <= 4'd0;
            sos_count <= 4'd0;
            sos_led   <= 1'b0;
        end else if (tick_4hz) begin
            case (sos_step)
                // S: dot dot dot
                0, 2, 4: begin
                    sos_led <= 1'b1;
                    if (sos_count == 4'd1) begin
                        sos_count <= 4'd0;
                        sos_step  <= sos_step + 4'd1;
                    end else
                        sos_count <= sos_count + 4'd1;
                end
                1, 3, 5: begin
                    sos_led <= 1'b0;
                    if (sos_count == 4'd1) begin
                        sos_count <= 4'd0;
                        sos_step  <= sos_step + 4'd1;
                    end else
                        sos_count <= sos_count + 4'd1;
                end
                // O: dash dash dash
                6, 8, 10: begin
                    sos_led <= 1'b1;
                    if (sos_count == 4'd3) begin
                        sos_count <= 4'd0;
                        sos_step  <= sos_step + 4'd1;
                    end else
                        sos_count <= sos_count + 4'd1;
                end
                7, 9, 11: begin
                    sos_led <= 1'b0;
                    if (sos_count == 4'd1) begin
                        sos_count <= 4'd0;
                        sos_step  <= sos_step + 4'd1;
                    end else
                        sos_count <= sos_count + 4'd1;
                end
                12: begin
                    sos_step  <= 4'd0;
                    sos_count <= 4'd0;
                end
                default: begin
                    sos_step  <= 4'd0;
                    sos_count <= 4'd0;
                    sos_led   <= 1'b0;
                end
            endcase
        end
    end

    //----------------------------------------
    // WARNING FLASHER (2 Hz)
    //----------------------------------------
    reg [24:0] warn_cnt;
    reg warn_led;

    always @(posedge MAX10_CLK1_50 or negedge reset_n) begin
        if (!reset_n) begin
            warn_cnt <= 25'd0;
            warn_led <= 1'b0;
        end else begin
            if (warn_cnt == 25'd25_000_000) begin
                warn_cnt <= 25'd0;
                warn_led <= ~warn_led;
            end else
                warn_cnt <= warn_cnt + 25'd1;
        end
    end

    //----------------------------------------
    // OUTPUT MUX (SOS > WARNING > normal tasks)
    //----------------------------------------
    always @(*) begin
        if (sos_mode)
            LEDR = {10{sos_led}};
        else if (warn_mode)
            LEDR = {5'b10101, 5'b01010} ^ {10{warn_led}};
        else if (is_blink_second)
            LEDR = t2_LEDRs;
        else if (is_task3)
            LEDR = t3_LEDRs;
        else
            LEDR = t4_LEDRs;
    end

endmodule

//------------------------ task2 ------------------------
// Blink all LEDs whenever seconds = 10,20,30,40,50
module task2(
    input wire clk,
    input wire reset_n,
    input wire enable,
    input wire tick,
    input wire [5:0] seconds,
    output reg [9:0] LEDRs
);
    always @(*) begin
        if (enable && (
            (seconds == 6'd10) ||
            (seconds == 6'd20) ||
            (seconds == 6'd30) ||
            (seconds == 6'd40) ||
            (seconds == 6'd50)))
            LEDRs = 10'b1111111111;
        else
            LEDRs = 10'b0000000000;
    end
endmodule

//------------------------ task3 ------------------------
// Running light left to right: LED0 -> LED9 -> LED0
// Pauses in place during SOS/warn (slow_count frozen, pos held)
module task3(
    input wire clk,
    input wire reset_n,
    input wire enable,
    input wire tick,
    input wire sos_mode,     // used as generic "pause" signal
    output reg [9:0] LEDRs
);

    reg [3:0] pos;
    reg [22:0] slow_count;

    wire step = (slow_count == 23'd5_000_000);

    always @(posedge clk or negedge reset_n) begin
        if (!reset_n)
            slow_count <= 23'd0;
        else if (!sos_mode) begin
            if (step)
                slow_count <= 23'd0;
            else
                slow_count <= slow_count + 23'd1;
        end
    end

    always @(posedge clk or negedge reset_n) begin
        if (!reset_n)
            pos <= 4'd0;
        else if (!sos_mode && enable && step) begin
            if (pos == 4'd9)
                pos <= 4'd0;
            else
                pos <= pos + 4'd1;
        end
    end

    always @(*) begin
        if (enable)
            LEDRs = (10'b0000000001 << pos);
        else
            LEDRs = 10'b0000000000;
    end

endmodule

//------------------------ task4 ------------------------
// Running light right to left: LED9 -> LED0 -> LED9 (bounce)
// Pauses in place during SOS/warn (slow_count frozen, pos held)
module task4(
    input wire clk,
    input wire reset_n,
    input wire enable,
    input wire tick,
    input wire sos_mode,     // used as generic "pause" signal
    output reg [9:0] LEDRs
);

    reg [3:0] pos;
    reg dir;
    reg [22:0] slow_count;

    wire step = (slow_count == 23'd5_000_000);

    always @(posedge clk or negedge reset_n) begin
        if (!reset_n)
            slow_count <= 23'd0;
        else if (!sos_mode) begin
            if (step)
                slow_count <= 23'd0;
            else
                slow_count <= slow_count + 23'd1;
        end
    end

    always @(posedge clk or negedge reset_n) begin
        if (!reset_n) begin
            pos <= 4'd9;
            dir <= 1'b0;
        end else if (!sos_mode && enable && step) begin
            if (dir == 1'b0) begin
                if (pos == 4'd0) begin
                    dir <= 1'b1;
                    pos <= pos + 4'd1;
                end else
                    pos <= pos - 4'd1;
            end else begin
                if (pos == 4'd9) begin
                    dir <= 1'b0;
                    pos <= pos - 4'd1;
                end else
                    pos <= pos + 4'd1;
            end
        end
    end

    always @(*) begin
        if (enable)
            LEDRs = (10'b0000000001 << pos);
        else
            LEDRs = 10'b0000000000;
    end

endmodule

//------------------------ clock divider ------------------------
module clock(
    input wire clk,
    input wire reset_n,
    output reg tick
);
    localparam integer DIV_COUNT = 50_000_000;
    reg [25:0] count;

    always @(posedge clk or negedge reset_n) begin
        if (!reset_n) begin
            count <= 26'd0;
            tick  <= 1'b0;
        end else begin
            if (count == DIV_COUNT - 1) begin
                count <= 26'd0;
                tick  <= 1'b1;
            end else begin
                count <= count + 26'd1;
                tick  <= 1'b0;
            end
        end
    end
endmodule

//------------------------ seven segment ------------------------
module sevenseg_driver(
    input wire [3:0] value,
    output reg [6:0] seg
);
    always @(*) begin
        case (value)
            4'h0: seg = 7'b1000000;
            4'h1: seg = 7'b1111001;
            4'h2: seg = 7'b0100100;
            4'h3: seg = 7'b0110000;
            4'h4: seg = 7'b0011001;
            4'h5: seg = 7'b0010010;
            4'h6: seg = 7'b0000010;
            4'h7: seg = 7'b1111000;
            4'h8: seg = 7'b0000000;
            4'h9: seg = 7'b0010000;
            default: seg = 7'b1111111;
        endcase
    end
endmodule
